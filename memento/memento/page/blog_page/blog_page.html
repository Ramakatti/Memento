Frappe Python Example

The following example demonstrates how a Frappe backend method can retrieve project information from a DocType.

Python 
import frappe

@frappe.whitelist()
def get_project_details(project_id):

    if not project_id:

        return {

            "success":
                False,

            "message":
                "Project ID is required"

        }


    project = frappe.get_doc(

        "My Projects",

        project_id

    )


    return {

        "name":
            project.name,

        "project_name":
            project.project_name,

        "status":
            project.status,

        "category":
            project.category,

        "priority":
            project.priority_level

    }

Frappe JavaScript Example

Frappe JavaScript can be used to retrieve DocType records and display information on the client side.

JavaScript 
function loadProject(projectId) {

    if (!projectId) {

        console.error(
            "Project ID is missing"
        );

        return;
    }


    frappe.db.get_doc(

        "My Projects",

        projectId

    ).then(function(project) {

        console.log(
            "Project:",
            project
        );


        console.log(
            "Name:",
            project.project_name
        );


        console.log(
            "Status:",
            project.status
        );


        console.log(
            "Category:",
            project.category
        );


    }).catch(
        function(error) {

            console.error(

                
                    "Failed to load project:"
                ,

                error

            );

        }
    );

}
